20:  * Único punto de entrada para modificar stock. Atómico, idempotente y sin stock negativo.
21:  */
22: final class RegisterMovement
23: {
24:     public function __construct(
25:         private readonly StockLedger $ledger,
26:         private readonly Transaction $transaction,
27:         private readonly EventBus $events,
28:     ) {}
29: 
30:     public function __invoke(NewMovement $movement): MovementRecorded
31:     {
32:         return $this->transaction->run(fn (): MovementRecorded => $this->execute($movement));
33:     }
34: 
35:     private function execute(NewMovement $m): MovementRecorded
36:     {
37:         // 1) El bloqueo del producto va PRIMERO: serializa reintentos concurrentes y el costo promedio.
38:         $product = $this->ledger->lockProduct($m->productId)
39:             ?? throw InventoryException::productNotFound($m->productId);
40: 
41:         // 2) Idempotencia.
42:         $existing = $this->ledger->findByIdempotencyKey($m->idempotencyKey);
43:         if ($existing !== null) {
44:             if ($existing->uuid !== $m->uuid) {
45:                 throw InventoryException::idempotencyKeyReused($m->idempotencyKey);
46:             }
47: 
48:             return new MovementRecorded($existing->id, $existing->uuid, true);
49:         }
50: 
51:         // 3) Ubicaciones válidas.
52:         $locations = $this->resolveLocations($m);
53: 
54:         // 4) Stock suficiente en el origen (salidas y transferencias).
55:         if ($m->fromLocationId !== null) {
56:             $available = $this->ledger->lockBalance($m->productId, $m->fromLocationId);
57:             if ($available->lessThan($m->quantity)) {
58:                 throw InventoryException::insufficientStock($locations[$m->fromLocationId]->code, $available, $m->quantity);
59:             }
60:         }
61: 
62:         // 5) Valorización.
63:         $recordedCost = $this->valuate($m, $product);
64: 
65:         // 6) Ledger + saldos (misma transacción).
66:         $movementId = $this->ledger->append($m, $recordedCost);
67:         $this->applyToBalances($m, $locations);
68: 
69:         $this->events->publish(new StockChanged($movementId, $m->productId, $m->type, $m->locationIds()));
70: 
71:         return new MovementRecorded($movementId, $m->uuid, false);
72:     }
73: 
74:     /** @return array<int, LocationState> */
75:     private function resolveLocations(NewMovement $m): array
76:     {
77:         $states = $this->ledger->locationStates($m->locationIds());
78: 
79:         foreach ($m->locationIds() as $id) {
80:             $state = $states[$id] ?? throw InventoryException::locationNotFound($id);
81: 
82:             if (! $state->isActive) {
83:                 throw InventoryException::locationInactive($state->code);
84:             }
85: 
86:             if ($state->isFrozen && ! $m->allowsFrozenLocations()) {
87:                 throw InventoryException::locationFrozen($state->code);
88:             }
89:         }
90: 
91:         return $states;
92:     }

94:     private function valuate(NewMovement $m, ProductSnapshot $product): UnitCost
95:     {
96:         if (! $m->type->requiresCost()) {
97:             return $product->costAvg; // salidas, mermas, ajustes y devoluciones valen al promedio vigente
98:         }
99: 
100:         $cost = $m->unitCost ?? throw InventoryException::invalid('Falta el costo unitario.');
101:         $newAverage = WeightedAverageCost::next(
102:             $this->ledger->totalQuantity($m->productId),
103:             $product->costAvg,
104:             $m->quantity,
105:             $cost,
106:         );
107:         $this->ledger->updateProductCost(
108:             $m->productId,
109:             $newAverage,
110:             $m->type === MovementType::Purchase ? $cost : null,
111:         );
112: 
113:         return $cost;
114:     }

118:     {
119:         if ($m->fromLocationId !== null && ! $this->ledger->decrease($m->productId, $m->fromLocationId, $m->quantity)) {
120:             // Defensa en profundidad: el chequeo previo ya debería haberlo impedido.
121:             $available = $this->ledger->lockBalance($m->productId, $m->fromLocationId);
122:             throw InventoryException::insufficientStock($locations[$m->fromLocationId]->code, $available, $m->quantity);
123:         }
