10: final class CsvExport
11: {
12:     /**
13:      * @param  list<string>  $headers
14:      * @param  iterable<list<scalar|null>>  $rows
15:      */
16:     public static function download(string $baseName, array $headers, iterable $rows): StreamedResponse
17:     {
18:         return response()->streamDownload(static function () use ($headers, $rows): void {
19:             $out = fopen('php://output', 'w');
20:             if ($out === false) {
21:                 return;
22:             }
23:             fwrite($out, "\xEF\xBB\xBF");
24:             fputcsv($out, $headers, ';', '"', '');
25:             foreach ($rows as $row) {
26:                 fputcsv($out, $row, ';', '"', '');
27:             }
28:             fclose($out);
29:         }, $baseName.'-'.now('America/Lima')->format('Y-m-d').'.csv', ['Content-Type' => 'text/csv; charset=UTF-8']);
30:     }
