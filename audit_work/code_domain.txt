2: 
3: declare(strict_types=1);
4: 
5: namespace App\Modules\Catalog\Domain;
6: 
7: /** Código de barras validado (dígito verificador GS1 para EAN-13, EAN-8 y UPC-A). */
8: final readonly class Barcode
9: {
10:     private function __construct(public string $value, public Symbology $symbology) {}
11: 
12:     public static function of(string $value, Symbology $symbology): self
13:     {
14:         $code = trim($value);
15: 
16:         $length = match ($symbology) {
17:             Symbology::Ean13 => 13,
18:             Symbology::Ean8 => 8,
19:             Symbology::UpcA => 12,
20:             default => null,
21:         };
22: 
23:         if ($length !== null) {
24:             if (preg_match('/^\d{'.$length.'}$/', $code) !== 1) {
25:                 throw CatalogException::invalid("El código {$symbology->value} debe tener {$length} dígitos.");
26:             }
27:             if (! self::hasValidCheckDigit($code)) {
28:                 throw CatalogException::invalid("El dígito verificador del código {$code} no es válido.");
29:             }
30:         } elseif (preg_match('/^[\x21-\x7E]{1,40}$/', $code) !== 1) {
31:             throw CatalogException::invalid('El código debe tener de 1 a 40 caracteres imprimibles, sin espacios.');
32:         }
33: 
34:         return new self($code, $symbology);
35:     }
36: 
37:     private static function hasValidCheckDigit(string $code): bool
38:     {
39:         $payload = substr($code, 0, -1);
40:         $sum = 0;
41:         $n = strlen($payload);
42: 
43:         for ($i = $n - 1; $i >= 0; $i--) {
44:             $weight = (($n - 1 - $i) % 2 === 0) ? 3 : 1; // el dígito más cercano al verificador pesa 3
45:             $sum += (int) $payload[$i] * $weight;
46:         }
47: 
48:         return ((10 - ($sum % 10)) % 10) === (int) substr($code, -1);
49:     }
50: }

51:     public static function normalizeName(string $name): string
52:     {
53:         $clean = trim((string) preg_replace('/\s+/u', ' ', $name));
54: 
55:         if (mb_strlen($clean) < 2 || mb_strlen($clean) > 200) {
56:             throw CatalogException::invalid('El nombre debe tener entre 2 y 200 caracteres.');
57:         }
58: 
59:         return $clean;
60:     }
61: 
62:     public static function normalizeDescription(?string $description): ?string
63:     {
64:         if ($description === null || trim($description) === '') {
65:             return null;
66:         }
67:         if (mb_strlen($description) > 2000) {
68:             throw CatalogException::invalid('La descripción no puede superar 2000 caracteres.');
69:         }
70: 
71:         return trim($description);
72:     }
73: 
74:     /**
75:      * @param  list<PresentationInput>  $input
76:      * @return list<PresentationInput>
77:      */
78:     private static function normalizePresentations(int $baseUnitId, array $input): array
79:     {
80:         $base = new PresentationInput($baseUnitId, Factor::of(1));
81:         $seen = [$baseUnitId => true];
82:         $out = [$base];
83:         $defaults = 0;
84: 
85:         foreach ($input as $p) {
86:             if ($p->unitId === $baseUnitId) {
87:                 if (! $p->factor->isOne()) {
88:                     throw CatalogException::invalid('La presentación de la unidad base debe tener factor 1.');
89:                 }
90:                 $out[0] = $p;
91:             } else {
92:                 if (isset($seen[$p->unitId])) {
93:                     throw CatalogException::invalid('Hay presentaciones repetidas para la misma unidad.');
94:                 }
95:                 $seen[$p->unitId] = true;
96:                 $out[] = $p;
97:             }
98:             $defaults += $p->isPurchaseDefault ? 1 : 0;
99:         }
100: 
101:         if ($defaults > 1) {
102:             throw CatalogException::invalid('Solo una presentación puede ser la de compra por defecto.');
103:         }
104: 
105:         return $out;
106:     }
107: 
108:     /**
109:      * @param  list<BarcodeInput>  $barcodes
110:      * @param  list<PresentationInput>  $presentations
111:      * @return list<BarcodeInput>
112:      */
113:     private static function normalizeBarcodes(array $barcodes, array $presentations): array
114:     {
115:         if ($barcodes === []) {
116:             return [];
117:         }
118: 
119:         $units = array_map(static fn (PresentationInput $p): int => $p->unitId, $presentations);
120:         $seen = [];
121:         $primaries = 0;
122: 
123:         foreach ($barcodes as $b) {
124:             if (isset($seen[$b->barcode->value])) {
125:                 throw CatalogException::invalid("El código {$b->barcode->value} está repetido en el formulario.");
126:             }
127:             $seen[$b->barcode->value] = true;
128: 
129:             if ($b->presentationUnitId !== null && ! in_array($b->presentationUnitId, $units, true)) {
130:                 throw CatalogException::invalid('Un código de barras referencia una presentación que no existe.');
131:             }
132:             $primaries += $b->isPrimary ? 1 : 0;
133:         }
134: 
135:         if ($primaries > 1) {
136:             throw CatalogException::invalid('Solo un código puede ser el principal.');
137:         }
138:         if ($primaries === 0) {
139:             $barcodes[0] = $barcodes[0]->asPrimary();
140:         }
141: 
142:         return $barcodes;
143:     }
144: 
145:     /**
146:      * @param  list<StockPolicy>  $policies
147:      * @return list<StockPolicy>
148:      */
149:     private static function normalizePolicies(array $policies): array
150:     {
151:         $seen = [];
152:         foreach ($policies as $p) {
153:             if (isset($seen[$p->warehouseId])) {
154:                 throw CatalogException::invalid('Hay más de una política de stock para el mismo almacén.');
155:             }
156:             $seen[$p->warehouseId] = true;
157:         }
158: 
159:         return $policies;
160:     }
161: }

15:         'contrasena1234', 'contraseña1234', 'password1234', 'administrador', 'administrador1',

1: <?php
2: 
3: declare(strict_types=1);
4: 
5: namespace App\Modules\Inventory\Domain;
6: 
7: /**
8:  * Costo promedio ponderado global por producto:
9:  *   nuevo = (Q_total x promedio_actual + q x costo) / (Q_total + q)
10:  * Redondeo half-up a 4 decimales.
11:  */
12: final class WeightedAverageCost
13: {
14:     public static function next(
15:         Quantity $currentQty,
16:         UnitCost $currentAvg,
17:         Quantity $incomingQty,
18:         UnitCost $incomingCost,
19:     ): UnitCost {
20:         $denominator = bcadd($currentQty->value, $incomingQty->value, 3);
21: 
22:         if (bccomp($denominator, '0', 3) === 0) {
23:             return $incomingCost;
24:         }
25: 
26:         $numerator = bcadd(
27:             bcmul($currentQty->value, $currentAvg->value, 7),
28:             bcmul($incomingQty->value, $incomingCost->value, 7),
29:             7,
30:         );
31: 
32:         $quotient = bcdiv($numerator, $denominator, 10);
33: 
34:         // bcadd trunca: sumar 0.00005 antes de truncar a 4 decimales equivale a half-up (valores >= 0).
35:         return UnitCost::of(bcadd($quotient, '0.00005', 4));
36:     }
37: }

7: /** Pedido de generación en lote: zona fija x pasillos x anaqueles x niveles. */
8: final readonly class LocationBatch
9: {
10:     public const MAX_PER_BATCH = 2000;
11: 
12:     public function __construct(
13:         public int $warehouseId,
14:         public int $zone,
15:         public int $aisleFrom,
16:         public int $aisleTo,
17:         public int $rackFrom,
18:         public int $rackTo,
19:         public int $levelFrom,
20:         public int $levelTo,
21:         public ?int $actorId = null,
22:         public ?string $maxCapacity = null,
23:     ) {
24:         foreach (['pasillos' => [$aisleFrom, $aisleTo], 'anaqueles' => [$rackFrom, $rackTo], 'niveles' => [$levelFrom, $levelTo]] as $label => [$from, $to]) {
25:             if ($from > $to) {
26:                 throw LocationException::invalid("El rango de {$label} es inválido: el inicial es mayor que el final.");
27:             }
28:         }
29: 
30:         if ($this->count() > self::MAX_PER_BATCH) {
31:             throw LocationException::invalid('Máximo '.self::MAX_PER_BATCH.' ubicaciones por lote (pidió '.$this->count().').');
32:         }

35:         }
36:     }

39:     {
40:         return ($this->aisleTo - $this->aisleFrom + 1) * ($this->rackTo - $this->rackFrom + 1) * ($this->levelTo - $this->levelFrom + 1);

48:             for ($rack = $this->rackFrom; $rack <= $this->rackTo; $rack++) {
49:                 for ($level = $this->levelFrom; $level <= $this->levelTo; $level++) {
50:                     $code = LocationCode::build($warehouseCode, $this->zone, $aisle, $rack, $level);

53:                         $this->maxCapacity === null ? null : bcadd($this->maxCapacity, '0', 3),

5: namespace App\Modules\Operations\Domain;

7: /** Lee el CSV de ventas del día que exporta la caja: columnas SKU (o código de barras) y cantidad. */
8: final class PosSalesParser
9: {
10:     public const MAX_ROWS = 20000;
11: 
12:     private const CODE_HEADERS = ['sku', 'codigo', 'cod', 'codigo_barras', 'barcode', 'ean', 'producto'];
13: 
14:     private const QTY_HEADERS = ['cantidad', 'cant', 'qty', 'unidades', 'quantity'];
15: 
16:     public static function parse(string $csv): PosParseResult
17:     {
18:         $csv = str_starts_with($csv, "\xEF\xBB\xBF") ? substr($csv, 3) : $csv;
19:         $lines = preg_split('/\r\n|\r|\n/', $csv) ?: [];
20: 
21:         $numbered = [];
22:         foreach ($lines as $i => $line) {
23:             if (trim($line) !== '') {
24:                 $numbered[] = [$i + 1, $line];
25:             }
26:         }
27:         if ($numbered === []) {
28:             throw OperationsException::invalid('El archivo está vacío.');
29:         }
30: 
31:         [, $headerLine] = $numbered[0];
32:         $delimiter = self::detectDelimiter($headerLine);
33:         $headers = array_map([self::class, 'normalizeHeader'], str_getcsv($headerLine, $delimiter, '"', ''));
34: 
35:         $codeIndex = self::indexOf($headers, self::CODE_HEADERS);
36:         $qtyIndex = self::indexOf($headers, self::QTY_HEADERS);
37:         if ($codeIndex === null || $qtyIndex === null) {
38:             throw OperationsException::invalid('El archivo debe tener las columnas "sku" (o código) y "cantidad" en la primera fila.');
39:         }
40: 
41:         $data = array_slice($numbered, 1);
42:         if (count($data) > self::MAX_ROWS) {
43:             throw OperationsException::invalid('El archivo supera el máximo de '.self::MAX_ROWS.' filas.');
44:         }
45: 
46:         $rows = [];
47:         $errors = [];
48:         foreach ($data as [$lineNumber, $line]) {
49:             $cells = str_getcsv($line, $delimiter, '"', '');
50:             $code = trim((string) ($cells[$codeIndex] ?? ''));
51:             $rawQty = trim((string) ($cells[$qtyIndex] ?? ''));
52: 
53:             if ($code === '') {
54:                 $errors[] = new ImportError($lineNumber, 'Falta el SKU o código del producto.');
55: 
56:                 continue;
57:             }
58:             if ($delimiter !== ',' && str_contains($rawQty, ',') && ! str_contains($rawQty, '.')) {
59:                 $rawQty = str_replace(',', '.', $rawQty); // coma decimal (Excel en español)
60:             }
61:             if (str_starts_with($rawQty, '-')) {
62:                 $errors[] = new ImportError($lineNumber, 'Cantidad negativa: las devoluciones no se importan aquí.');
63: 
64:                 continue;
65:             }
66:             if (preg_match('/^\d{1,9}(\.\d{1,3})?$/', $rawQty) !== 1 || bccomp($rawQty, '0', 3) <= 0) {
67:                 $errors[] = new ImportError($lineNumber, "Cantidad inválida: '{$rawQty}'.");
68: 
69:                 continue;
70:             }
71: 
72:             $rows[] = new PosSaleRow($lineNumber, $code, bcadd($rawQty, '0', 3));

76:     }

83:         return (string) array_key_first($counts);
84:     }

90: 
91:         return trim(str_replace([' ', '-'], '_', $h), "\"' ");
92:     }

96:      * @param  list<string>  $accepted
97:      */

99:     {

FILE app\Modules\Suppliers\Domain\Ruc.php
1: <?php
2: 
3: declare(strict_types=1);
4: 
5: namespace App\Modules\Suppliers\Domain;
6: 
7: /** RUC peruano de 11 dígitos con dígito verificador (módulo 11 de SUNAT). */
8: final readonly class Ruc implements \Stringable
9: {
10:     private const WEIGHTS = [5, 4, 3, 2, 7, 6, 5, 4, 3, 2];
11: 
12:     private const PREFIXES = ['10', '15', '16', '17', '20'];
13: 
14:     private function __construct(public string $value) {}
15: 
16:     public static function of(string $value): self
17:     {
18:         $ruc = trim($value);
19: 
20:         if (preg_match('/^\d{11}$/', $ruc) !== 1) {
21:             throw SupplierException::invalid('El RUC debe tener 11 dígitos.');
22:         }
23:         if (! in_array(substr($ruc, 0, 2), self::PREFIXES, true)) {
24:             throw SupplierException::invalid('El RUC debe empezar con 10, 15, 16, 17 o 20.');
25:         }
26: 
27:         $sum = 0;
28:         foreach (self::WEIGHTS as $i => $weight) {
29:             $sum += (int) $ruc[$i] * $weight;
30:         }
31:         $check = 11 - ($sum % 11);
32:         $check = $check === 10 ? 0 : ($check === 11 ? 1 : $check);
33: 
34:         if ($check !== (int) $ruc[10]) {
35:             throw SupplierException::invalid('El RUC no es válido (dígito verificador incorrecto).');
36:         }
37: 
38:         return new self($ruc);
39:     }
40: 
41:     public function __toString(): string
42:     {
43:         return $this->value;
44:     }
45: }

FILE app\Modules\Suppliers\Domain\SupplierFields.php
1: <?php
2: 
3: declare(strict_types=1);
4: 
5: namespace App\Modules\Suppliers\Domain;
6: 
7: /** Validación y normalización compartida por altas y ediciones. */
8: final class SupplierFields
9: {
10:     /** @return array{business_name: ?string, contact_name: ?string, phone: ?string, email: ?string, address: ?string} */
11:     public static function clean(?string $businessName, ?string $contactName, ?string $phone, ?string $email, ?string $address, ?int $leadTimeDays, bool $nameRequired = true): array
12:     {
13:         $name = $businessName === null ? null : trim((string) preg_replace('/\s+/u', ' ', $businessName));
14:         if ($nameRequired && ($name === null || mb_strlen($name) < 2)) {
15:             throw SupplierException::invalid('La razón social debe tener al menos 2 caracteres.');
16:         }
17:         if ($name !== null && mb_strlen($name) > 160) {
18:             throw SupplierException::invalid('La razón social no puede superar 160 caracteres.');
19:         }
20: 
21:         $contact = self::blankToNull($contactName);
22:         if ($contact !== null && mb_strlen($contact) > 120) {
23:             throw SupplierException::invalid('El contacto no puede superar 120 caracteres.');
24:         }
25: 
26:         $phone = self::blankToNull($phone);
27:         if ($phone !== null && preg_match('/^[0-9+()\s-]{6,30}$/', $phone) !== 1) {
28:             throw SupplierException::invalid('El teléfono no es válido.');
29:         }
30: 
31:         $email = self::blankToNull($email);
32:         $email = $email === null ? null : mb_strtolower($email);
33:         if ($email !== null && (filter_var($email, FILTER_VALIDATE_EMAIL) === false || mb_strlen($email) > 160)) {
34:             throw SupplierException::invalid('El correo no es válido.');
35:         }
36: 
37:         $address = self::blankToNull($address);
38:         if ($address !== null && mb_strlen($address) > 255) {
39:             throw SupplierException::invalid('La dirección no puede superar 255 caracteres.');
40:         }
41: 
42:         if ($leadTimeDays !== null && ($leadTimeDays < 0 || $leadTimeDays > 365)) {
43:             throw SupplierException::invalid('El plazo de entrega debe estar entre 0 y 365 días.');
44:         }
45: 
46:         return ['business_name' => $name, 'contact_name' => $contact, 'phone' => $phone, 'email' => $email, 'address' => $address];
47:     }
48: 
49:     private static function blankToNull(?string $value): ?string
50:     {
51:         $v = $value === null ? null : trim($value);
52: 
53:         return $v === null || $v === '' ? null : $v;
54:     }
55: }
