18:           JOIN locations l ON l.id = sb.location_id
19:           JOIN warehouses w ON w.id = l.warehouse_id
20:           JOIN products p ON p.id = sb.product_id AND p.deleted_at IS NULL
21:           LEFT JOIN categories c ON c.id = p.category_id
22:          WHERE sb.quantity > 0
23:          GROUP BY p.id, p.sku, p.name, p.cost_avg, c.name, w.id, w.name;
24: 
25:         -- Rotación y clasificación ABC (últimos 90 días, por valor vendido al costo):
26:         -- A = los que suman el 80 % del valor, B = hasta el 95 %, C = el resto, SIN_VENTAS = no se vendió nada.
27:         CREATE VIEW v_report_abc AS
28:         WITH sales AS (
29:             SELECT product_id, SUM(quantity) AS qty_90d, SUM(quantity * COALESCE(unit_cost, 0)) AS value_90d
30:               FROM stock_movements
31:              WHERE type = 'SALIDA_VENTA' AND created_at >= now() - INTERVAL '90 days'
32:              GROUP BY product_id
33:         ), stock AS (
34:             SELECT product_id, SUM(quantity) AS qty FROM stock_balances GROUP BY product_id
35:         ), ranked AS (
36:             SELECT s.product_id, s.qty_90d, s.value_90d,
37:                    SUM(s.value_90d) OVER (ORDER BY s.value_90d DESC, s.product_id) AS cum_value,
38:                    SUM(s.value_90d) OVER () AS total_value
39:               FROM sales s WHERE s.value_90d > 0
40:         )
41:         SELECT p.id AS product_id, p.sku, p.name AS product_name, c.name AS category_name,
42:                COALESCE(r.qty_90d, 0) AS qty_90d, COALESCE(r.value_90d, 0) AS value_90d,
43:                CASE WHEN r.product_id IS NULL THEN NULL ELSE ROUND(100 * r.value_90d / NULLIF(r.total_value, 0), 2) END AS share_pct,
44:                CASE WHEN r.product_id IS NULL THEN 'SIN_VENTAS'
45:                     WHEN (r.cum_value - r.value_90d) < 0.80 * r.total_value THEN 'A'
46:                     WHEN (r.cum_value - r.value_90d) < 0.95 * r.total_value THEN 'B'
47:                     ELSE 'C' END AS abc_class,
48:                COALESCE(st.qty, 0) AS stock,
49:                CASE WHEN COALESCE(r.qty_90d, 0) > 0 THEN ROUND(COALESCE(st.qty, 0) / (r.qty_90d / 90.0), 0) END AS cover_days,
50:                p.cost_avg
51:           FROM products p
52:           LEFT JOIN ranked r ON r.product_id = p.id
53:           LEFT JOIN stock st ON st.product_id = p.id
54:           LEFT JOIN categories c ON c.id = p.category_id
55:          WHERE p.deleted_at IS NULL AND p.status = 'ACTIVE';
56: 
57:         -- Mermas y ajustes negativos, con motivo, responsable y valor al costo.
58:         CREATE VIEW v_report_waste AS
59:         SELECT m.id, m.created_at, m.type, p.sku, p.name AS product_name, l.code AS location_code,
60:                r.name AS reason_name, m.quantity, COALESCE(m.unit_cost, 0) AS unit_cost,
61:                m.quantity * COALESCE(m.unit_cost, 0) AS value,
62:                u.name AS user_name, a.name AS approver_name, m.document_ref, m.notes
63:           FROM stock_movements m
64:           JOIN products p ON p.id = m.product_id
65:           JOIN locations l ON l.id = m.from_location_id
66:           LEFT JOIN movement_reasons r ON r.id = m.reason_id
67:           JOIN users u ON u.id = m.user_id
68:           LEFT JOIN users a ON a.id = m.approved_by
69:          WHERE m.type IN ('MERMA', 'AJUSTE_NEGATIVO');
70:         SQL);
71:     }

429: -- ---------------------------------------------------------------------
430: -- 8. VISTAS, FUNCIONES Y REPORTES
431: -- ---------------------------------------------------------------------
432: -- Cada movimiento produce 1 o 2 "efectos" (delta) sobre ubicaciones.
433: CREATE VIEW v_movement_effects AS
434: SELECT id AS movement_id, product_id, to_location_id AS location_id, quantity AS delta, type, created_at
435:   FROM stock_movements WHERE to_location_id IS NOT NULL
436: UNION ALL
437: SELECT id, product_id, from_location_id, -quantity, type, created_at
438:   FROM stock_movements WHERE from_location_id IS NOT NULL;
439: 
440: CREATE VIEW v_kardex AS
441: SELECT e.movement_id, e.product_id, e.location_id, l.warehouse_id, e.type, e.delta, e.created_at,
442:        SUM(e.delta) OVER (PARTITION BY e.product_id, e.location_id
443:                           ORDER BY e.created_at, e.movement_id) AS running_balance
444:   FROM v_movement_effects e JOIN locations l ON l.id = e.location_id;
445: 
446: CREATE VIEW v_product_stock AS
447: SELECT b.product_id, l.warehouse_id, SUM(b.quantity) AS quantity
448:   FROM stock_balances b JOIN locations l ON l.id = b.location_id
449:  GROUP BY b.product_id, l.warehouse_id;
450: 
451: -- Reconciliación: debe devolver 0 filas. Si devuelve filas, se abre alerta CRITICAL.
452: CREATE OR REPLACE FUNCTION fn_reconcile_ledger()
453: RETURNS TABLE (p_product_id BIGINT, p_location_id BIGINT, p_balance NUMERIC, p_ledger NUMERIC, p_diff NUMERIC)
454: LANGUAGE sql STABLE AS $$
455:   SELECT COALESCE(b.product_id, l.product_id), COALESCE(b.location_id, l.location_id),
456:          COALESCE(b.quantity, 0), COALESCE(l.q, 0), COALESCE(b.quantity, 0) - COALESCE(l.q, 0)
457:     FROM stock_balances b
458:     FULL JOIN (SELECT product_id, location_id, SUM(delta) AS q
459:                  FROM v_movement_effects GROUP BY product_id, location_id) l
460:       ON l.product_id = b.product_id AND l.location_id = b.location_id
461:    WHERE COALESCE(b.quantity, 0) <> COALESCE(l.q, 0);
462: $$;
463: 
464: -- Rotación y cobertura (REFRESH MATERIALIZED VIEW CONCURRENTLY cada noche)
465: CREATE MATERIALIZED VIEW mv_product_rotation_90d AS
466: SELECT p.id AS product_id,
467:        COALESCE(SUM(m.quantity) FILTER (WHERE m.type = 'SALIDA_VENTA'), 0) AS qty_sold_90d,
468:        COALESCE(SUM(m.quantity * COALESCE(m.unit_cost, p.cost_avg)) FILTER (WHERE m.type = 'SALIDA_VENTA'), 0) AS value_sold_90d,
469:        (SELECT MAX(created_at) FROM stock_movements x WHERE x.product_id = p.id) AS last_movement_at
470:   FROM products p
471:   LEFT JOIN stock_movements m ON m.product_id = p.id AND m.created_at >= now() - INTERVAL '90 days'
472:  WHERE p.deleted_at IS NULL
473:  GROUP BY p.id;
474: CREATE UNIQUE INDEX uq_mv_rotation ON mv_product_rotation_90d (product_id);
